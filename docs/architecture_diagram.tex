% ── Styles ────────────────────────────────────────────────────
\tikzset{
  % Generic rounded box; border colour passed as argument
  nodebox/.style={
    rectangle, draw=#1, fill=white, line width=1.2pt,
    rounded corners=8pt, inner sep=10pt,
    minimum width=3.6cm, minimum height=1.2cm,
    align=center, font=\sffamily
  },
  nodebox/.default=colHost,
  nFL/.style  ={nodebox=colFL},
  nTrk/.style ={nodebox=colTrk},
  nObs/.style ={nodebox=colObs},
  nHost/.style={nodebox=colHost},
  % Database cylinder
  db/.style={
    cylinder, draw=colFL, fill=white, line width=1.0pt,
    shape border rotate=90, aspect=0.28,
    minimum width=2.2cm, minimum height=1.1cm,
    align=center, font=\sffamily\small
  },
  % Side information boxes
  infobox/.style={
    rectangle, draw=#1, fill=white, line width=1.0pt,
    rounded corners=8pt, inner sep=10pt,
    align=left, font=\sffamily\small
  },
  infobox/.default=colFL,
  % Arrow styles
  arr/.style={
    ->, draw=colArrow, line width=0.9pt,
    >=Stealth, rounded corners=5pt
  },
  dasharr/.style={
    ->, draw=colTrk, line width=0.8pt,
    dashed, dash pattern=on 5pt off 2.5pt,
    >=Stealth, rounded corners=8pt
  },
  grpcarr/.style={
    <->, draw=colFL, line width=1.0pt,
    >=Stealth, rounded corners=5pt
  },
  % Unidirectional FL arrow (colFL coloured, used in pairs)
  flarr/.style={
    ->, draw=colFL, line width=1.0pt,
    >=Stealth, rounded corners=5pt
  },
  % Dashed group enclosure
  boundary/.style={
    draw=colBound, dashed, line width=0.9pt,
    rounded corners=14pt, inner sep=20pt
  },
  % Edge label helper
  el/.style={
    fill=white, inner sep=2pt,
    font=\sffamily\scriptsize, text=black!60
  }
}
